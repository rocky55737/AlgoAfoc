# 코드
```
#include <iostream>
#include <vector>

bool dp[2001][2001];

using namespace std;

int main()
{
  ios::sync_with_stdio(false);
  cin.tie(nullptr);

  int N;
  cin >> N;
  for (int i = 0; i <= N; i++)
  {
    for (int j = 0; j <= N; j++)
    {
      if (i == 1 && j != 0)
      {
        dp[i][j] = true;
      }
      else
      {
        dp[i][j] = false;
      }
    }
  }
  vector<int> numbers_vector(N + 1, 0);
  numbers_vector[0] = 0;
  for (int i = 1; i <= N; i++)
  {
    cin >> numbers_vector[i];
    if (numbers_vector[i] == numbers_vector[i - 1])
    {
      dp[2][i - 1] = true;
    }
    if (i > 2 && numbers_vector[i] == numbers_vector[i - 2])
    {
      dp[3][i - 2] = true;
    }
  }

  for (int i = 3; i <= N; i++)
  {
    for (int j = 1; j <= N - i + 1; j++)
    {
      if (numbers_vector[j] == numbers_vector[j + i - 1] && dp[i - 2][j + 1])
      {
        dp[i][j] = true;
      }
    }
  }

  int M;
  cin >> M;
  for (int i = 0; i < M; i++)
  {
    int start, end;
    cin >> start >> end;
    if (dp[end - start + 1][start])
    {
      cout << "1\n";
    }
    else
    {
      cout << "0\n";
    }
  }
  return 0;
}
```

# 풀이 과정
동적 프로그래밍으로 풀었다.  
먼저 모든 경우에서 펠린드롬인지 아닌지를 확인해두었다.  
이를 확인하는 방법은 다음과 같다.  
  
1. 길이가 1이면 무조건 펠린드롬이다.  
2. 길이가 2이면 숫자가 같으면 펠린드롬이다.  
3. 길이가 3보다 크면 양 끝이 같고 그 속이 펠린드롬이면 펠린드롬이다.
